🎯 روش‌های تغییر تم:

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1️⃣  تغییر از طریق URL (سریع‌ترین روش)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

در آدرس مرورگر، پارامتر ?theme= را اضافه کنید:

  🌸 بهار:     ?theme=spring
  ☀️ تابستان:  ?theme=summer
  🍂 پاییز:     ?theme=autumn
  ❄️ زمستان:   ?theme=winter

مثال:
  http://localhost/Velora/public/?theme=winter
  http://localhost/Velora/public/login?theme=spring


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
2️⃣  تغییر از منوی سایت
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

در Navbar بالای صفحه، روی دکمه "تم" کلیک کنید و تم دلخواه 
را انتخاب کنید.


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
3️⃣  تغییر توسط ادمین (برای همه کاربران)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. وارد پنل ادمین شوید:
   http://localhost/Velora/public/admin

2. از منو، "تنظیمات تم" را انتخاب کنید

3. گزینه Manual را انتخاب کرده و تم دلخواه را ست کنید


━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
4️⃣  تغییر مستقیم از فایل تنظیمات
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

فایل: config/theme.php را باز کنید و ویرایش کنید:

<?php
return [
    'mode' => 'manual',        // automatic یا manual
    'admin_selected_theme' => 'winter'  // spring, summer, autumn, winter
];

⚠️ توجه: پس از تغییر، مرورگر را رفرش کنید (Ctrl + F5)


═══════════════════════════════════════════════════════════════════
📋 لیست تم‌های موجود:
═══════════════════════════════════════════════════════════════════

  🌸 spring   - بهار   (سبز، صورتی، رنگ‌های بهاری)
  ☀️ summer   - تابستان (آبی، زرد، رنگ‌های تابستانی)
  🍂 autumn   - پاییز   (نارنجی، قهوه‌ای، رنگ‌های پاییزی)
  ❄️ winter   - زمستان (آبی روشن، سفید، یخی)
